  <script>
    if (
      sessionStorage.getItem("modelsReadonly") !== "1" ||
      sessionStorage.getItem("modelsRole") !== "readonly"
    ) {
      window.location.href = "/admin/login.html";
    }

    let SCHEDULES = [];

    const DAY_NAMES = [
      "Sunday",
      "Monday",
      "Tuesday",
      "Wednesday",
      "Thursday",
      "Friday",
      "Saturday"
    ];

    function logout() {
      sessionStorage.removeItem("modelsReadonly");
      sessionStorage.removeItem("modelsRole");
      window.location.href = "/admin/login.html";
    }

    function getStartOfWeek(date = new Date()) {
      const d = new Date(date);
      d.setHours(0, 0, 0, 0);

      const day = d.getDay(); // Sunday=0, Monday=1...
      const diff = day === 0 ? -6 : 1 - day; // move to Monday
      d.setDate(d.getDate() + diff);

      return d;
    }

    function getEndOfWeek(startOfWeek) {
      const d = new Date(startOfWeek);
      d.setDate(d.getDate() + 6);
      d.setHours(23, 59, 59, 999);
      return d;
    }

    function formatDateISO(date) {
      const year = date.getFullYear();
      const month = String(date.getMonth() + 1).padStart(2, "0");
      const day = String(date.getDate()).padStart(2, "0");
      return `${year}-${month}-${day}`;
    }

    function formatShortDate(iso) {
      const [year, month, day] = iso.split("-");
      return `${month}/${day}/${year}`;
    }

    function getWeekDates() {
      const start = getStartOfWeek(new Date());
      const days = [];

      for (let i = 0; i < 7; i++) {
        const d = new Date(start);
        d.setDate(start.getDate() + i);

        days.push({
          dateObj: d,
          iso: formatDateISO(d),
          dayName: DAY_NAMES[d.getDay()]
        });
      }

      return days;
    }

    function getWeekInfo() {
      const start = getStartOfWeek(new Date());
      const end = getEndOfWeek(start);

      return {
        start,
        end,
        startISO: formatDateISO(start),
        endISO: formatDateISO(end)
      };
    }

    function updateWeekLabels() {
      const { startISO, endISO } = getWeekInfo();
      const label = `${formatShortDate(startISO)} - ${formatShortDate(endISO)}`;

      document.getElementById("weekBadge").textContent = `Current week: ${label}`;
      document.getElementById("weekRangeText").textContent = label;
    }

    function normalizeSchedules(raw) {
      return (raw || [])
        .filter(item => item && item.date && item.start && item.modelName)
        .map(item => ({
          modelName: String(item.modelName).trim(),
          theme: String(item.theme || "").trim(),
          date: String(item.date).trim(),
          start: String(item.start).trim()
        }));
    }

    function filterCurrentWeek(items) {
      const { startISO, endISO } = getWeekInfo();
      return items.filter(item => item.date >= startISO && item.date <= endISO);
    }

    function sortSchedules(items) {
      return [...items].sort((a, b) => {
        if (a.date === b.date) {
          return a.start.localeCompare(b.start);
        }
        return a.date.localeCompare(b.date);
      });
    }

    function groupSchedulesByDay(items) {
      const grouped = {};
      const weekDays = getWeekDates();

      weekDays.forEach(day => {
        grouped[day.iso] = [];
      });

      items.forEach(item => {
        if (grouped[item.date]) {
          grouped[item.date].push(item);
        }
      });

      Object.keys(grouped).forEach(date => {
        grouped[date].sort((a, b) => a.start.localeCompare(b.start));
      });

      return grouped;
    }

    function renderBoard() {
      const contentArea = document.getElementById("contentArea");
      const weekDays = getWeekDates();
      const weekItems = sortSchedules(filterCurrentWeek(SCHEDULES));
      const grouped = groupSchedulesByDay(weekItems);

      document.getElementById("totalStreams").textContent = weekItems.length;

      const uniqueModels = new Set(
        weekItems.map(item => item.modelName.toLowerCase())
      );
      document.getElementById("totalModels").textContent = uniqueModels.size;

      const boardHTML = weekDays.map(day => {
        const entries = grouped[day.iso] || [];

      const content = entries.length
  ? entries.map(item => `
      <div class="stream-card">
        <div class="stream-time">${item.start}</div>
        <div class="stream-model">${escapeHtml(item.modelName)}</div>
        <div class="stream-theme">
          <strong>Theme:</strong> ${escapeHtml(item.theme || "Not provided")}
        </div>
      </div>
    `).join("")
  : `<div class="empty-day">No livestreams scheduled</div>`;

        return `
          <section class="day-column">
            <div class="day-header">
              <div class="day-name">${day.dayName}</div>
              <div class="day-date">${formatShortDate(day.iso)}</div>
            </div>
            <div class="day-content">
              ${content}
            </div>
          </section>
        `;
      }).join("");

      contentArea.innerHTML = `
        <div class="board-wrapper">
          <div class="board">
            ${boardHTML}
          </div>
        </div>
      `;
    }

    function escapeHtml(value) {
      return value
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
    }

    function loadSchedules() {
      updateWeekLabels();

      document.getElementById("contentArea").innerHTML =
        '<div class="loading-box">Loading weekly schedule...</div>';

      fetch("/api/model-schedule")
        .then(res => {
          if (!res.ok) {
            throw new Error(`HTTP ${res.status}`);
          }
          return res.json();
        })
        .then(data => {
          SCHEDULES = normalizeSchedules(data);
          renderBoard();
        })
        .catch(err => {
          console.error("Error loading schedule:", err);
          document.getElementById("contentArea").innerHTML =
            '<div class="error-box">Failed to load the weekly schedule.</div>';
        });
    }

    loadSchedules();
  </script>
